// Invariant: Only one leader per term
pub async fn start_election(&mut self) -> Result<Role, ConsensusError> {
    self.current_term += 1;
    self.voted_for = Some(self.node_id);
    let mut votes_received = 1;

    // Broadcast RequestVote RPCs in parallel
    for peer in &self.peers {
        if let Ok(vote) = peer.request_vote(self.current_term).await {
            if vote.term > self.current_term {
                self.step_down(vote.term);
                return Ok(Role::Follower);
            }
        }
    }
}
